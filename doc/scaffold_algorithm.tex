\begin{algorithm}[t]
\DontPrintSemicolon
\caption{Scalar-field construction and surface extraction (Main Pipeline)}
\label{alg:scalarfield_main}
\KwIn{seed set $\mathcal{S}$; container SDF $\Psi_\Omega$; grid resolution $(N_x,N_y,N_z)$;
      iso-level $c_{\mathrm{glob}}$; openness $\tau\in[0,1]$;
      global background stretch $\mathbf{S}_{\mathrm{bg}}$ and orientation $(\hat{\mathbf{d}}_{\mathrm{bg}}, \theta_{\mathrm{bg}})$;
      local anisotropy sources $\mathcal{A}=\{(\mathbf{o}_m,\mathbf{R}_m,\mathbf{S}_m,\sigma_m)\}$;
      background blend weight $w_b>0$;
      Taubin smoothing parameters $(N_{\mathrm{iter}}, \lambda, \mu)$, $0<\lambda<-\mu$;
      optional thickness modulator $(\Psi,\rho,t_{\min},t_{\max})$.}
\KwOut{watertight triangle mesh $\mathcal{M}$ of the iso-surface $\Phi=c_{\mathrm{glob}}$.}
\BlankLine
$\mathbf{C}_m \leftarrow \mathbf{R}_m^{\!\top} \mathbf{S}_m^2 \mathbf{R}_m$ for every source $m$;\quad
$\mathbf{C}_{\mathrm{bg}} \leftarrow \mathbf{R}_{\mathrm{bg}}^{\!\top} \mathbf{S}_{\mathrm{bg}}^2 \mathbf{R}_{\mathrm{bg}}$
  \tcp*{covariance (inverse-metric) tensors}
\eIf{$\mathcal{A} = \emptyset$}{
  Warp all seeds: $\mathbf{s}_i \leftarrow \mathbf{S}_{\mathrm{bg}}^{-1}\mathbf{R}_{\mathrm{bg}}(\mathbf{s}_i - \mathbf{c}_\Omega)$
    \tcp*{$\lVert\mathbf{S}^{-1}\mathbf{R}\,\boldsymbol\delta\rVert^2 = \boldsymbol\delta^{\!\top}\mathbf{C}_{\mathrm{bg}}^{-1}\boldsymbol\delta$}
  $K \leftarrow 3$\;
}{
  $K \leftarrow \min\bigl(\max(\lceil 3\kappa^2\rceil,\,12),\,|\mathcal{S}|\bigr)$,\quad
  $\kappa = s_{\max}/s_{\min}$ over all stretches \tcp*{candidate budget}
}
$\mathcal{T} \leftarrow \mathrm{KdTree}(\mathcal{S})$;
\quad initialise $\Phi(i,j,k) \leftarrow \Phi_\infty$\;
\BlankLine
$\mathcal{B}_{\mathrm{narrow}} \leftarrow \mathrm{ClassifyNarrowBand}(\Psi_\Omega, \text{margin})$
  \tcp*{coarse Lipschitz bound; exact SDF only near $\partial\Omega$}
\BlankLine
\ForAll(\tcp*[h]{parallel}){voxels $(i,j,k)$}{
  $\mathbf{x} \leftarrow$ world-space position of $(i,j,k)$\;
  \If{$\mathcal{B}_{\mathrm{narrow}}(\mathbf{x}) = \mathrm{OUTSIDE}$}{
    $\Phi(i,j,k) \leftarrow c_{\mathrm{glob}}+1$;\quad \textbf{continue} \tcp*{far outside the domain}
  }
  $\Phi(i,j,k) \leftarrow \mathrm{EvaluateVoxel}(\mathbf{x}, \mathcal{T}, \mathcal{A}, \mathbf{C}_{\mathrm{bg}}, w_b, K, \tau, c_{\mathrm{glob}}, \dots)$
    \tcp*{Algorithm~\ref{alg:evaluate_voxel}}
}
\BlankLine
$\Phi \leftarrow \mathrm{TaubinSmoothField}\bigl(\Phi,\, N_{\mathrm{iter}}, \lambda, \mu\bigr)$
  \tcp*{$\lambda|\mu$ band-pass; per-axis weights $\propto 1/h_a^2$}
\ForAll{voxels $(i,j,k)$}{
  $\Phi(i,j,k) \leftarrow \max\bigl(\Phi(i,j,k),\;\Psi_\Omega(\mathbf{x})+c_{\mathrm{glob}}\bigr)$ \tcp*{boundary clamp (cached SDF)}
}
$\Phi \leftarrow \mathrm{KeepLargestComponent}(\Phi,\,c_{\mathrm{glob}})$\;
$\Phi \leftarrow \mathrm{SealBoundary}(\Phi)$\;
$\mathcal{M} \leftarrow \mathrm{LewinerMC}(\Phi,\,c_{\mathrm{glob}})$ \tcp*{topologically consistent MC33}
\Return $\mathcal{M}$\;
\end{algorithm}

\begin{algorithm}[t]
\DontPrintSemicolon
\caption{Sub-routine: $\mathrm{EvaluateVoxel}$}
\label{alg:evaluate_voxel}
\KwIn{position $\mathbf{x}$; KD-tree $\mathcal{T}$; sources $\mathcal{A}$; background covariance $\mathbf{C}_{\mathrm{bg}}$;
      background weight $w_b$; candidate budget $K$; openness $\tau$; base iso-level $c_{\mathrm{glob}}$.}
\KwOut{scalar field value at $\mathbf{x}$.}
\BlankLine
$c_{\mathrm{loc}} \leftarrow c_{\mathrm{glob}}$\;
\If{thickness modulation enabled}{
  $c_{\mathrm{loc}} \leftarrow \rho\bigl(\,|\Psi(\mathbf{x})|;\,t_{\min},t_{\max}\bigr)$\;
}
\BlankLine
\eIf{$\mathcal{A} \neq \emptyset$}{
  \ForEach{source $m \in \mathcal{A}$}{
    $w_m \leftarrow \max\Bigl(0,\;
        e^{-\lVert \mathbf{x} - \mathbf{o}_m \rVert^2 / 2\sigma_m^2} - e^{-9/2}\Bigr)$
      \tcp*{vanishes continuously at the $3\sigma_m$ support}
  }
  $\mathbf{C}(\mathbf{x}) \leftarrow
     \dfrac{\sum_m w_m \mathbf{C}_m \;+\; w_b\,\mathbf{C}_{\mathrm{bg}}}
           {\sum_m w_m \;+\; w_b}$
     \tcp*{partition-of-unity blend}
  $\mathbf{M}(\mathbf{x}) \leftarrow \mathbf{C}(\mathbf{x})^{-1}$ \tcp*{local Riemannian metric}
  $\mathcal{K} \leftarrow \mathcal{T}.\mathrm{knn}(\mathbf{x},\,K)$ \tcp*{Euclidean candidates}
  re-rank $\mathcal{K}$ by $d_M(\mathbf{x},\mathbf{p})^2 = (\mathbf{x}-\mathbf{p})^{\!\top}\mathbf{M}(\mathbf{x})(\mathbf{x}-\mathbf{p})$;\;
  keep the three smallest $\bigl\{(\mathbf{p}_k,\,d_k)\bigr\}_{k=1}^{3}$;\quad
  $\nabla d_k \leftarrow \mathbf{M}(\mathbf{x})(\mathbf{x}-\mathbf{p}_k)\,/\,d_k$\;
}{
  $\mathbf{x}' \leftarrow \mathbf{S}_{\mathrm{bg}}^{-1}\mathbf{R}_{\mathrm{bg}}(\mathbf{x}-\mathbf{c}_\Omega)$
    \tcp*{same warp as the seeds}
  $\bigl\{(\mathbf{p}'_k,\,d_k)\bigr\}_{k=1}^{3} \leftarrow \mathcal{T}.\mathrm{knn}(\mathbf{x}',\,3)$
    \tcp*{uniform warped Euclidean query}
  $\nabla d_k \leftarrow \mathbf{R}_{\mathrm{bg}}^{\!\top}\mathbf{S}_{\mathrm{bg}}^{-1}(\mathbf{x}'-\mathbf{p}'_k)\,/\,d_k$
    \tcp*{chain rule back to world space}
}
\BlankLine
$v_{\mathrm{wall}} \leftarrow d_2 - d_1$;\quad $\nabla v_{\mathrm{wall}} \leftarrow \nabla d_2 - \nabla d_1$\;
$v_{\mathrm{strut}} \leftarrow d_3 - d_1$;\quad $\nabla v_{\mathrm{strut}} \leftarrow \nabla d_3 - \nabla d_1$\;
\BlankLine
$v \leftarrow (1-\tau)v_{\mathrm{wall}} + \tau v_{\mathrm{strut}}$
  \tcp*{$\tau{=}0$: closed walls (foam), $\tau{=}1$: open struts (lattice)}
$\nabla v \leftarrow (1-\tau)\nabla v_{\mathrm{wall}} + \tau \nabla v_{\mathrm{strut}}$\;
\BlankLine
$\widetilde{v} \leftarrow v$\;
\If{$\lVert\nabla v\rVert > \epsilon$}{
  $\widetilde{v} \leftarrow 2\,v\,/\,\lVert\nabla v\rVert$ \tcp*{scale-invariant (approximate) wall distance}
}
\Return $\widetilde{v} - c_{\mathrm{loc}} + c_{\mathrm{glob}}$\;
\end{algorithm}
